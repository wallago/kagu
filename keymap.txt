// Kagu keymap: the source of truth for config/boards/shields/kagu/kagu.keymap
// and, through keymap-drawer, keymap/keymap.svg. Edit this file, then run
// `just keymap`.
// The format comes from denniskempin/dubu36.
//
// Every `overlay` and `layer` block is a grid of the 36 keys: three rows of ten
// keys, then the six thumb keys. Each whitespace separated cell is one key:
//
//   LABEL             tap only
//   TAP/HOLD          taps TAP, holds HOLD (a modifier or a layer)
//   TAP/HOLD:FLAVOR   ... using an explicit hold-tap flavor
//   _                 nothing (&none), or "no hold" when used as the hold
//   TRNS              falls through to the layer below (&trans)
//
// FLAVOR picks the hold-tap behavior. A layer hold always uses &lt. A modifier
// hold uses &mt, or with `hm` the positional home row mods from
// keymap/template.keymap (&hml on the left half, &hmr on the right).
//
// A layer inherits the holds of every overlay listed after its name, unless the
// cell defines its own hold. Use `LABEL/_` to drop an inherited hold. On a cell
// with no tap, the inherited hold becomes a plain key press.
//
// `CTL_X` sends Ctrl+X, `GUI_X` Super+X, `CS_X` Ctrl+Shift+X. `HYP_X` sends X with Ctrl+Alt+Shift+Super held, for
// niri bindings. `TAP|SHIFTED` taps TAP, or SHIFTED (without Shift) while
// Shift is held.
//
// Labels that clash with the `/`, `:` or `_` syntax are escaped with a
// backslash: `\/`, `\:`, `\_`, and `\\` for the backslash key itself.
//
// A `combos` block is `KEY KEY -> RESULT`. The header names the layers the
// chords fire on (`combos: base`), and may add `idle=MS` to require that long
// since the last key press. The two keys are named by what they tap on the
// first layer listed.
//
// A cell that repeats its label, `LSHFT/LSHFT` or `sym/sym`, is one-shot on
// tap and momentary on hold. `layer name + MOD` holds MOD for as long as the
// layer is held from a hold-tap, so every key on it is sent with MOD.
//
// The first layer is the default one. Layer names become the BASE, RSE, ...
// defines in kagu.keymap.

// Layout after dubu36: modifiers used in shortcuts sit on the home row,
// while shift and the layers used in fluent writing sit on the thumbs, which
// keeps hold-tap timing out of the way while typing.

// Home row mods, pinky to index: Shift, Ctrl, Alt, Super, mirrored on the
// right. The right hand holds left Alt, since right Alt is AltGr on Linux. A
// mod only fires when the next key is on the other hand or a thumb.
overlay left_mods
  _         _         _        _        _          _  _  _  _  _
  LSHFT:hm  LCTRL:hm  LALT:hm  LGUI:hm  _          _  _  _  _  _
  _         _         _        _        _          _  _  _  _  _
                      _        _        _          _  _  _

overlay right_mods
  _  _  _  _  _          _  _        _        _         _
  _  _  _  _  _          _  RGUI:hm  LALT:hm  RCTRL:hm  RSHFT:hm
  _  _  _  _  _          _  _        _        _         _
        _  _  _          _  _        _

// Colemak-DH. Bksp, Enter (Shift on hold) and
// one-shot lwr on the left thumbs; one-shot rse, Space and hyp on the right.
layer base: left_mods, right_mods
  Q      W        F      P     B          J        L     U      Y     '
  A      R        S      T     G          M        N     E      I     O
  Z      X        C      D     V          K        H     ,      .     \/
                  BSPC   RET/LSHFT    lwr/lwr     rse/rse  SPC   _/hyp

// Numbers on the left, navigation on the right, with three Hyper keys for
// niri below the arrows. 0 on the middle left thumb. Bluetooth profiles 0-2
// down the inner column and BT_CLR on the inner thumb; the outer thumb
// switches to the game layer. No home row holds here, so digits and arrows
// repeat when held.
layer rse
  _      7      8      9      BT_0       HOME    WORD_L   UP       WORD_R   END
  _      4      5      6      BT_1       BACK    LEFT     DOWN     RIGHT    FWD
  _      1      2      3      BT_2       TAB_L   HYP_[    HYP_\\   HYP_]    TAB_R
                TO_GAME 0     BT_CLR     _       _        _

// Symbols, with no home row holds.
layer lwr
  ~      ^      @      $      %          &      \/     \\     PIPE   `
  ESC    <      [      (      {          -      \_     \:     ;      #
  *      >      ]      )      }          +      =      ?      !      "
                _      _      _          STAB   TAB    _

// Hyper (Ctrl+Alt+Shift+Super) shortcuts for niri, held from the right
// outer thumb; niri binds each one to an action. Left: workspaces 1-9 and 0
// as a numpad, move column (top) and focus column (home) on the outer
// columns, terminal (T) and launcher (D) below, consume/expel ([ ]) on the
// thumbs. Right: arrows move the column (left, right) and the window inside
// it (up, down); C center, Q close, O overview, F maximize, R width preset,
// V float, - and = width, W tabbed column, M fullscreen, P screenshot.
layer hyp
  HYP_LEFT  HYP_7  HYP_8  HYP_9  HYP_RIGHT      HYP_C      HYP_Q      HYP_UP     HYP_O      HYP_F
  HYP_H     HYP_4  HYP_5  HYP_6  HYP_L          HYP_R      HYP_LEFT   HYP_DOWN   HYP_RIGHT  HYP_V
  HYP_T     HYP_1  HYP_2  HYP_3  HYP_D          HYP_-      HYP_=      HYP_W      HYP_M      HYP_P
                   HYP_[  HYP_0  HYP_]          _          _          TRNS

// One-handed gaming (Dofus) with the right hand on the mouse, entered from
// rse (outer left thumb). Every left key is a plain press, no hold-tap, so
// nothing waits on a timer: spells 1-5 on top and Ctrl+1-5 (second spell bar)
// on the home row, ~ below them, plain Shift on the outer thumb, F1 on the middle thumb
// ends the turn. Hold Shift for 6-0 and Ctrl+6-0 instead. The right half
// falls through to base. Hold the inner thumb for menu.
layer game
  1|6          2|7          3|8          4|9          5|0            TRNS  TRNS  TRNS  TRNS  TRNS
  CTL_1|CTL_6  CTL_2|CTL_7  CTL_3|CTL_8  CTL_4|CTL_9  CTL_5|CTL_0    TRNS  TRNS  TRNS  TRNS  TRNS
  ~            _            _            _            _              TRNS  TRNS  TRNS  TRNS  TRNS
                            LSHFT        F1           MENU           TRNS  TRNS  TRNS
